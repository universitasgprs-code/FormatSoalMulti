📊 9. Integrasi Google Spreadsheet & Drive (Kirim Nilai Otomatis)

Masukkan URL Web App dari Google Apps Script untuk menerima hasil ujian siswa secara otomatis ke Google Spreadsheet. Anda juga dapat menyimpan informasi lokasi folder Google Drive tempat spreadsheet berada.

✅ URL Google Apps Script berhasil disimpan di perangkat ini!
📌 PENTING: Klik tombol "💾 Download HTML Terbaru" di bawah lalu ganti file di GitHub Pages/hosting Anda agar URL otomatis terpasang di seluruh HP/perangkat siswa tanpa perlu copy-paste lagi.
💡 Panduan Pembuatan Google Apps Script:
  1. Buat Google Spreadsheet baru di Google Drive Anda.
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus kode lama dan tempel kode berikut: 
    function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000); // Tunggu antrean maksimal 15 detik agar aman antar pengiriman
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);

    // Otomatis buat header kolom sesuai dengan hasil nilai siswa jika sheet masih kosong
    if (sheet.getLastRow() == 0) {
      sheet.appendRow([
        "Timestamp",
        "Nama Lengkap",
        "Nomor Absen",
        "Kelas",
        "Mata Pelajaran",
        "Nilai Ujian",
        "Nilai Maksimal",
        "KKM",
        "Status Kelulusan",
        "Jumlah Benar",
        "Total Soal",
        "Jumlah Pelanggaran",
        "Auto Submit",
        "Waktu Selesai"
      ]);
    }

    // PENCEGAHAN DATA GANDA (Deduplikasi di Spreadsheet):
    // Cek apakah data siswa dengan nama, nomor, kelas, mapel sudah ada di sheet
    var lastRow = sheet.getLastRow();
    if (lastRow > 1 && data.nama && data.kelas && data.mataPelajaran && data.nama !== "Uji Coba Sistem") {
      var range = sheet.getRange(2, 2, lastRow - 1, 4).getValues(); // Ambil Kolom Nama(B), No(C), Kelas(D), Mapel(E)
      var targetNama = String(data.nama).trim().toLowerCase();
      var targetNomor = String(data.nomor || '').trim().toLowerCase();
      var targetKelas = String(data.kelas).trim().toLowerCase();
      var targetMapel = String(data.mataPelajaran).trim().toLowerCase();

      for (var r = 0; r < range.length; r++) {
        var rowNama = String(range[r][0] || '').trim().toLowerCase();
        var rowNomor = String(range[r][1] || '').trim().toLowerCase();
        var rowKelas = String(range[r][2] || '').trim().toLowerCase();
        var rowMapel = String(range[r][3] || '').trim().toLowerCase();

        if (rowNama === targetNama && rowNomor === targetNomor && rowKelas === targetKelas && rowMapel === targetMapel) {
          // Data sudah tercatat sebelumnya di spreadsheet, jangan buat baris ganda
          return ContentService.createTextOutput(JSON.stringify({result: "duplicate_skipped", message: "Data sudah ada di spreadsheet"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }

    sheet.appendRow([
      data.waktuSelesai || new Date().toLocaleString('id-ID'),
      data.nama || "",
      data.nomor || "",
      data.kelas || "",
      data.mataPelajaran || "",
      data.nilai !== undefined ? data.nilai : 0,
      data.nilaiMaksimal || 100,
      data.kkm || 75,
      data.statusKelulusan || "",
      data.benar !== undefined ? data.benar : 0,
      data.totalSoal || 0,
      data.pelanggaran !== undefined ? data.pelanggaran : 0,
      data.autoSubmit ? "Ya" : "Tidak",
      data.waktuSelesai || ""
    ]);

    SpreadsheetApp.flush();
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
  4. Klik Terapkan (Deploy) > Penerapan baru (New deployment).
  5. Pilih jenis: Aplikasi web (Web app).
  6. Setel "Siapa yang memiliki akses" (Who has access) ke: Siapa saja (Anyone).
  7. Klik Terapkan (Deploy), lalu berikan izin akses:
    • Jika muncul peringatan "Google hasn't verified this app", klik Advanced (Tingkat lanjut) di kiri bawah.
    • Klik Go to ... (unsafe) / Lanjutkan ke ... (tidak aman).
    • Klik Allow / Izinkan.
    • (Peringatan ini normal karena skrip dibuat sendiri dan belum didaftarkan sebagai aplikasi publik oleh Google).
